\section{Executive Summary}
% ============================================================================
CPG\_DISHA is a comprehensive event-based payment collection and management system designed for institutions to efficiently manage student payments through multiple bank integrations. The system orchestrates complex payment workflows including payment request creation, multi-bank payment processing, automatic reconciliation, refund handling, and settlement tracking.

\subsection{Key Capabilities}
\begin{itemize}
  \item \textbf{Event Management}: System heads create events and define payment collection requirements.
  \item \textbf{Flexible Payment Requests}: Support for fixed-amount, variable-amount, and recurring payments.
  \item \textbf{ICICI Bank Integration}: Full lifecycle payment processing (initiation, status verification, refunds, settlements).
  \item \textbf{Automatic Reconciliation}: Daily scheduled reconciliation of pending/failed payments with ICICI.
  \item \textbf{Duplicate Detection}: Duplicate payments are queued as refund records in the `Refunds` collection for manual or asynchronous processing; the system does not call external refund APIs automatically.\
  \item \textbf{Duplicate Indicators}: Queued duplicate payments are surfaced in system-head and admin transaction history views and flagged in user receipts to aid reconciliation and merchant review.
  \item \textbf{External Payment Links}: Public payment links for non-registered users.
  \item \textbf{Role-Based Access Control}: Three-tier permission model (admin, system\_head, user).
  \item \textbf{Settlement Tracking}: Daily ICICI settlement history synchronization and reporting.
\end{itemize}

\subsection{Authentication and User Management}

\subsubsection{FR-AUTH-001: User Login}
\begin{itemize}
  \item Users submit email and password
  \item System verifies against bcrypt hashed passwords (SALT\_ROUNDS=10)
  \item Upon success: returns JWT token + user object (minus password)
  \item Token valid for 15 minutes
  \item Endpoint: POST /auth/login
  \item File: backend/routes/auth.js
\end{itemize}

\subsubsection{FR-AUTH-002: User Signup}
\begin{itemize}
  \item Self-signup available for `user` role only (config: SELF\_SIGNUP\_ROLES)
  \item Required fields: name, email, password (min 8 chars), roll\_no (for users)
  \item Email validation: normalized to lowercase
  \item Roll number validation: uppercase, whitespace stripped
  \item Password hashing: bcrypt with SALT\_ROUNDS=10
  \item Duplicate email check: MongoDB unique index on email
  \item Endpoint: POST /auth/signup
  \item File: backend/routes/auth.js
\end{itemize}

\subsubsection{FR-AUTH-003: Password Change}
\begin{itemize}
  \item Requires valid JWT token
  \item Old password verified via bcrypt.compare
  \item New password min 8 characters
  \item Updates passwordHash and mustChangePassword flag
  \item Endpoint: POST /auth/change-password
  \item File: backend/routes/auth.js
\end{itemize}

\subsubsection{FR-AUTH-004: Get Current User}
\begin{itemize}
  \item Retrieves authenticated user details
  \item Password hash excluded from response
  \item Endpoint: GET /auth/me
  \item File: backend/routes/auth.js
\end{itemize}

\subsubsection{FR-EVENT-004: Mark Event Complete}

\subsection{Payment Request Management}

\subsubsection{FR-PAYMENT-001: Create One-Time Payment Request}
\begin{itemize}
  \item Type: one\_time
  \item Batch structure: multiple rollNo/amount entries
  \item Required: at least one valid entry, timeToLive (ISO date)
  \item Validation: no duplicate roll numbers within batch
  \item Bank selection: validates against enabled banks
  \item Output: creates Payment\_Request records, returns batch view
  \item Endpoint: POST /events/:eventId/payment-requests (type=one\_time)
  \item File: backend/routes/create-payment-request.js
\end{itemize}

\subsubsection{FR-PAYMENT-002: Create Recurring One-Time Payments}
\begin{itemize}
  \item Optional flag: isRecurring=true on one-time creation
  \item Required: intervalValue (positive number), intervalUnit (days|months)
  \item Creates Recurring\_Templates record with nextExecutionDate
  \item Template status: active
  \item Stored for daily scheduler execution
  \item Endpoint: POST /events/:eventId/payment-requests (isRecurring=true)
  \item File: backend/routes/create-payment-request.js
\end{itemize}

\subsubsection{FR-PAYMENT-003: Create Fixed Payment Request}
\begin{itemize}
  \item Type: fixed
  \item isAmountFixed flag: if true, amount required and must be > 0
  \item if false: amount submitted later by user (variable)
  \item Single record per request (not batched)
  \item Endpoint: POST /events/:eventId/payment-requests (type=fixed)
  \item File: backend/routes/create-payment-request.js
\end{itemize}

\subsubsection{FR-PAYMENT-004: Get Latest Payment Request}
\begin{itemize}
  \item Returns most recent payment request for event
  \item For one-time: reconstructs entries from batch
  \item Endpoint: GET /events/:eventId/payment-requests/latest
  \item File: backend/routes/create-payment-request.js
\end{itemize}

\subsubsection{FR-PAYMENT-005: Delete Fixed Payment Request}
\begin{itemize}
  \item Only system\_head who created can delete
  \item Endpoint: DELETE /events/fixed-payment-requests/:paymentRequestId
  \item File: backend/routes/delete-payment-request.js
\end{itemize}

\subsubsection{FR-PAYMENT-006: Delete One-Time Payment Request}

\subsubsection{FR-USER-PAYMENT-004: Initiate Payment Sale}
\begin{itemize}
  \item User selects payment request and bank

  \item Creates Payment\_Processed record with status='pending'

\subsubsection{FR-USER-PAYMENT-005: Verify Payment Status}
\begin{itemize}

  \item Updates PaymentRequest status to success/failed

  \item Endpoint: POST /user-payments/verify-status
  \item File: backend/routes/user-payments.js
\end{itemize}

\end{itemize}

\subsection{Security Requirements}

\subsubsection{NFR-SEC-001: Authentication}
\begin{itemize}
  \item JWT with 15-minute expiry
  \item Token claims: sub (user ID), role, email
  \item All protected routes require valid token via Authorization header (Bearer scheme)
  \item File: backend/middleware/auth.js
\end{itemize}

\subsubsection{NFR-SEC-002: Authorization}
\begin{itemize}
  \item Role-based access control: admin, system\_head, user
  \item RBAC enforced via requireRole middleware
  \item Data isolation: system\_heads only see own events/payments/links
  \item Users only see payments for their roll number
\end{itemize}

\subsubsection{NFR-SEC-003: Password Security}
\begin{itemize}
  \item Minimum 8 characters
  \item Bcrypt hashing with SALT\_ROUNDS=10
  \item Passwords never stored in plaintext
  \item Password hash never returned in API responses
\end{itemize}

\subsubsection{NFR-SEC-004: Data Validation}
\begin{itemize}
  \item Email normalization: lowercase, trim
  \item Roll number normalization: uppercase, whitespace stripped
  \item Amount validation: must be finite positive number
  \item Date validation: ISO format parsing
  \item Enum validation: roles, statuses, transaction types
  \item Max length checks on text fields
\end{itemize}

\subsubsection{NFR-SEC-005: ICICI Communication Security}
\begin{itemize}
  \item HMAC-SHA256 secure hash verification (sorted keys concatenated)
  \item Support for separate HMAC secrets for requests and responses: `ICICI_HMAC_SECRET` is used for outgoing request signing; when present `ICICI_HMAC_SECRET_RESPONSE` is used to verify incoming responses (falls back to `ICICI_HMAC_SECRET` if not set)
  \item Request and response packet integrity verified
  \item Hash mismatch handling: logs detailed diagnostics
  \item Fallback hash verification using multiple strategies (sorted, insertion order, skip falsy)
  \item File: backend/routes/bank-payment/icici.js
\end{itemize}

\subsubsection{NFR-SEC-006: Sensitive Data Protection}
\begin{itemize}
  \item Passwords: bcrypt hashed, never logged
  \item ICICI merchant credentials: stored in Banks collection, accessed via field lookups
  \item JWT secret: environment variable, not committed to repo
  \item HMAC secret: environment variable (ICICI\_HMAC\_SECRET)
  \item Request/response packets: logged for debugging (in production, consider PII masking)
\end{itemize}

\subsubsection{NFR-SEC-007: CORS and Headers}
\begin{itemize}
  \item CORS enabled: accepts requests from frontend origin
  \item Content-Type: application/json for API responses
  \item No sensitive headers exposed
\end{itemize}

  \item Partial failures: scheduler continues if one task fails
  \item File: backend/scheduler.js

\subsubsection{Database Layer: db.js}
Centralized MongoDB accessors:
\begin{itemize}
  \item Collection getters with ready-state checking (throws 503 if not ready)
  \item Specialized query functions (by ID, by email, by role, etc.)

export function requireAuth(req, res, next) {
  const authHeader = req.headers.authorization || "";
  const token = authHeader.startsWith("Bearer ") ? authHeader.slice(7) : "";
  if (!token) {
    return res.status(401).json({ message: "Missing access token" });
  }
  try {
    req.auth = jwt.verify(token, JWT_SECRET);
    return next();
  } catch {
    return res.status(401).json({ message: "Invalid or expired token" });
  }
}

export function requireRole(...allowedRoles) {
  return (req, res, next) => {
    if (!req.auth || !allowedRoles.includes(req.auth.role)) {
      return res.status(403).json({ message: "Forbidden" });
    }
    return next();
  };
}

\subsection{ICICI Bank Integration}

\subsubsection{Secure Hash Verification}
The system uses HMAC-SHA256 with sorted field values:
\begin{lstlisting}
function buildSecureHashFromSortedValues(packet) {
  const sortedKeys = Object.keys(packet).sort();
  const concatenatedValues = sortedKeys.map((key) => String(packet[key] ?? "")).join("");
  return crypto.createHmac("sha256", ICICI_HMAC_SECRET)
    .update(concatenatedValues).digest("hex");
}
\end{lstlisting}

\subsubsection{Sale Initiation Flow}
\begin{enumerate}
  \item Build request packet (amount, currencyCode=356, merchantId, aggregatorID, etc.)
  \item Compute secureHash from sorted values
  \item POST to ICICI\_INITIATE\_SALE\_URL
  \item Verify response secureHash
  \item Extract tranCtx from response
  \item Build paymentURL: ICICI\_AUTH\_REDIRECT\_URL?tranCtx=...
  \item Return to frontend
\end{enumerate}

\subsubsection{Status Check Flow}
\begin{enumerate}
  \item Build packet with merchantTxnNo or tranCtx
  \item Compute secureHash
  \item POST to ICICI\_STATUS\_CHECK\_URL
  \item Parse response (handles both JSON and URL-encoded)
  \item Verify secureHash (with multiple fallback strategies)
  \item Extract txnStatus and description
  \item Return status (success/failed/pending)
\end{enumerate}

\subsubsection{Refund Flow}
\begin{enumerate}
  \item When a duplicate successful payment is detected, create a refund record in the `Refunds` collection containing merchant and student details, original transaction id, amount and `status: "initiated"`
  \item Persist the refund record and do not call any external refund API automatically; external refund processing (manual or asynchronous worker) is out of scope for the immediate queued action
  \item The refund record stores request/response packets (when available), responseCode, respDescription and a `hashVerified` flag for later verification
  \item Update payment records with an alert indicating a queued refund and include refund metadata in history views
\end{enumerate}

\subsection{Scheduler and Automatic Reconciliation}

\subsubsection{Cron Schedule}
Runs daily at 00:00 IST (Asia/Kolkata timezone):
\begin{lstlisting}
cron.schedule("0 0 * * *", async () => {
  await executeRecurringTemplates();
  await retryPendingHashVerifications();
  await reconcilePendingAndFailedTransactions();
  await runIciciSettlementSync();
}, { scheduled: true, timezone: "Asia/Kolkata" });
\end{lstlisting}

\subsubsection{Recurring Template Execution}
\begin{enumerate}
  \item Fetch all active templates with nextExecutionDate <= today
  \item For each template:
    \begin{enumerate}
      \item Verify event still exists
      \item Calculate new TTL (original duration + today)
      \item Generate one-time payment requests from entries
      \item Calculate next execution date
      \item Update template
    \end{enumerate}
\end{enumerate}

\subsubsection{Hash Verification Retry}
\begin{enumerate}
  \item Find all pending records with pendingHashVerificationRetry=true
  \item For each record:
    \begin{enumerate}
      \item Call checkIciciSaleStatus
      \item If hash now verifies: update status (success/failed)
      \item Update payment request accordingly
    \end{enumerate}
\end{enumerate}

\subsubsection{Reconciliation Logic}
\begin{enumerate}
  \item Fetch all pending/failed payment records (non-external)
  \item For each record:
    \begin{enumerate}
      \item Check if exceeded retry limit (5 daily checks for pending, 2 days for failed)
      \item Call checkIciciSaleStatus
      \item Handle transitions:
      \begin{itemize}
        \item pending -> pending: increment check counter
        \item pending -> failed: max retries exceeded, append alert
        \item pending -> success: verify duplicate, process refund if needed
        \item failed -> success: append alert, process duplicate refund
        \item failed -> failed: continue retrying
      \end{itemize}
      \item Update Payment\_Processed and related Payment\_Request
    \end{enumerate}
\end{enumerate}

\subsubsection{Duplicate Detection and Refund}
When a payment transitions to success, the scheduler checks:
\begin{enumerate}
  \item Get all successful payments for same event+rollNo
  \item If count >= 2: newer payment is duplicate
  \item Check if refund already initiated
  \item Create a refund record in the `Refunds` collection (do not call external refund APIs automatically) with merchant and student details and `status: "initiated"`
  \item Append alert to payment record indicating refund queued
  \item Track refund status (initiated/success/failed) via the Refunds collection
\end{enumerate}

\subsubsection{Settlement Synchronization}
Queries ICICI for payment settlement status:
\begin{enumerate}
  \item Attempt summary query (SETTLEMENTSUMMARY)
  \item If success: record total settled amount and transaction count
  \item If failed: fallback to per-transaction status checks
  \item For each successful payment from previous IST day:
    \begin{enumerate}
      \item Call checkIciciSettlementStatus
      \item Collect settlement details (settled amount, status, timestamp)
    \end{enumerate}
  \item Upsert Settlement\_History record with results
\end{enumerate}

\section{Security Specifications}
% ============================================================================

\subsection{Authentication Security}
\begin{itemize}
  \item JWT tokens expire after 15 minutes
  \item Tokens contain: sub (user ID), role, email
  \item Authorization header required: ``Bearer <token>''
  \item Invalid/expired tokens return 401
  \item Frontend detects 401 and clears local auth state
\end{itemize}

\subsection{Authorization Security}
\begin{itemize}
  \item Role-based middleware enforces access control
  \item Data ownership checks: system\_head can only access own events
  \item User can only see payments for their roll number
  \item Admin can see all data but system\_heads cannot see peer data
\end{itemize}

\subsection{Password Security}
\begin{itemize}
  \item Minimum 8 characters enforced on signup/password-change
  \item Bcrypt with SALT\_ROUNDS=10 (default from bcryptjs)
  \item Passwords never logged or returned in API responses
  \item Password reset not implemented (must change password after first login)
\end{itemize}

\subsection{Input Validation and Sanitization}
\begin{itemize}
  \item Email: normalized (lowercase, trimmed)
  \item Roll number: uppercase, whitespace removed
  \item Amounts: parsed as numbers, must be finite and positive
  \item Dates: ISO format parsing with validation
  \item Enums: checked against allowed values
  \item Strings: max length validation on key fields
  \item Arrays: duplicate checks where necessary
\end{itemize}

\subsection{ICICI Communication Security}
\begin{itemize}
  \item HMAC-SHA256 hash verification on all ICICI responses

\subsection{HTTP Status Codes}
\begin{itemize}
  \item 200 OK: Successful GET/PATCH
  \item 201 Created: Successful POST creation
  \item 400 Bad Request: Validation failure, missing fields
  \item 401 Unauthorized: Missing or invalid token
  \item 403 Forbidden: Insufficient permissions
  \item 404 Not Found: Resource doesn't exist
  \item 409 Conflict: Duplicate resource, constraint violation
  \item 410 Gone: Resource disabled (external link)
  \item 500 Internal Server Error: Unhandled server error
  \item 502 Bad Gateway: External API failure (ICICI)
  \item 503 Service Unavailable: Database unavailable
\end{itemize}

\subsection{Global Error Handler}
\begin{lstlisting}
app.use((error, _req, res, _next) => {
  console.error(error);
  const status = Number(error?.status) || 500;
  if (status === 503) {
    return res.status(503).json({ message: "Database unavailable. Please retry shortly." });
  }
  return res.status(500).json({ message: "Internal server error" });
});
\end{lstlisting}

\subsection{Database Connection Retry}
\begin{itemize}
  \item Automatic retry with exponential backoff
  \item Initial delay: 3 seconds, capped at 30 seconds
  \item Attempts logged with timestamps
  \item Server continues to listen while connecting
\end{itemize}

\subsection{Hash Verification Fallback}
ICICI response verification attempts multiple strategies:
\begin{enumerate}
  \item Sorted alphabetically (standard method)
  \item Sorted with falsy values skipped
  \item Insertion order (in case ICICI uses response order)
\end{enumerate}

\subsection{Settlement Sync Fallback}
If settlement summary fails:
\begin{enumerate}
  \item Falls back to per-transaction status checks
  \item Processes all successful payments from previous day
  \item Records partial settlement status
  \item Continues scheduler on failure
\end{enumerate}

function buildSecureHashFromSortedValues(packet) {
  const sortedKeys = Object.keys(packet).sort();
  const concatenatedValues = sortedKeys.map((key) => String(packet[key] ?? "")).join("");
  return crypto
    .createHmac(ICICI_HMAC_ALGO || "sha256", ICICI_HMAC_SECRET)
    .update(concatenatedValues)
    .digest("hex");
}

// frontend/src/lib/api.js
export async function apiRequest(path, options = {}, token) {
  const headers = {
    "Content-Type": "application/json",
    ...(options.headers || {}),
  };
  if (token) {
    headers.Authorization = `Bearer ${token}`;
  }
  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers,
  });
  const payload = await response.json().catch(() => ({}));
  if (!response.ok) {
    if (response.status === 401 && token && typeof window !== "undefined") {
      window.dispatchEvent(
        new CustomEvent(AUTH_EXPIRED_EVENT, {
          detail: { path, message: payload.message || "Invalid or expired token" }
        })
      );
    }
    throw new Error(payload.message || "Request failed");
  }
  return payload;
}
